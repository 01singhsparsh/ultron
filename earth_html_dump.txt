// ── INIT THREE.JS ─────────────────────────────────────────────────
function initThree() {
  scene    = new THREE.Scene();
  clock    = new THREE.Clock();
  renderer = new THREE.WebGLRenderer({
    canvas: document.getElementById('mainCanvas'),
    antialias: true,
    logarithmicDepthBuffer: true,
  });
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  renderer.setSize(window.innerWidth, window.innerHeight);
  renderer.shadowMap.enabled = true;
  renderer.toneMapping = THREE.ACESFilmicToneMapping;
  renderer.toneMappingExposure = 1.2;

  camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.01, 10000);
  camera.position.set(0, 0, camState.distance);

  window.addEventListener('resize', () => {
    camera.aspect = window.innerWidth / window.innerHeight;
    camera.updateProjectionMatrix();
    renderer.setSize(window.innerWidth, window.innerHeight);
  });
}

// ── STARS ─────────────────────────────────────────────────────────
function createStars() {
  const geo = new THREE.BufferGeometry();
  const count = 12000;
  const pos = new Float32Array(count * 3);
  const col = new Float32Array(count * 3);
  for (let i = 0; i < count; i++) {
    const theta = Math.random() * Math.PI * 2;
    const phi   = Math.acos(2 * Math.random() - 1);
    const r     = 800 + Math.random() * 200;
    pos[i*3]   = r * Math.sin(phi) * Math.cos(theta);
    pos[i*3+1] = r * Math.sin(phi) * Math.sin(theta);
    pos[i*3+2] = r * Math.cos(phi);
    // Slightly varied colors
    const t = Math.random();
    col[i*3]   = 0.8 + t * 0.2;
    col[i*3+1] = 0.85 + t * 0.1;
    col[i*3+2] = 0.9 + t * 0.1;
  }
  geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
  geo.setAttribute('color',    new THREE.BufferAttribute(col, 3));
  const mat = new THREE.PointsMaterial({
    size: 0.5, vertexColors: true, transparent: true, opacity: 0.9,
    sizeAttenuation: true,
  });
  stars = new THREE.Points(geo, mat);
  scene.add(stars);
}

// ── NEBULA BACKGROUND ─────────────────────────────────────────────
function createNebula() {
  // Simple large transparent sphere with subtle nebula texture
  const geo = new THREE.SphereGeometry(900, 32, 32);
  const canvas = document.createElement('canvas');
  canvas.width = 1024; canvas.height = 512;
  const ctx = canvas.getContext('2d');

    map: tex, side: THREE.BackSide,
    transparent: true, opacity: 0.6,
  });
  scene.add(new THREE.Mesh(geo, mat));
}

// ── EARTH ─────────────────────────────────────────────────────────
function createEarth() {
  earthGroup = new THREE.Group();
  scene.add(earthGroup);

  // Earth sphere — use canvas texture for fallback then try real imagery
  const R = 1.0;
  const geo = new THREE.SphereGeometry(R, 128, 128);

  // Generate procedural Earth texture
  const earthCanvas = document.createElement('canvas');
  earthCanvas.width = 2048; earthCanvas.height = 1024;
  const ec = earthCanvas.getContext('2d');
  drawProceduralEarth(ec, 2048, 1024);

  const earthTex = new THREE.CanvasTexture(earthCanvas);

  // Try loading real texture
  const loader = new THREE.TextureLoader();
  loader.crossOrigin = 'anonymous';

  // Bump / specular generated
  const bumpCanvas = document.createElement('canvas');
  bumpCanvas.width = 1024; bumpCanvas.height = 512;
  const bc = bumpCanvas.getContext('2d');
  drawBumpMap(bc, 1024, 512);
  const bumpTex = new THREE.CanvasTexture(bumpCanvas);

  const mat = new THREE.MeshPhongMaterial({
    map:         earthTex,
    bumpMap:     bumpTex,
    bumpScale:   0.05,
    specularMap: bumpTex,
    specular:    new THREE.Color(0.2, 0.2, 0.4),
    shininess:   25,
  });

  earth = new THREE.Mesh(geo, mat);
  earthGroup.add(earth);

  // Clouds layer
  const cloudCanvas = document.createElement('canvas');
  cloudCanvas.width = 2048; cloudCanvas.height = 1024;
  const cc = cloudCanvas.getContext('2d');
  drawClouds(cc, 2048, 1024);
  const cloudTex = new THREE.CanvasTexture(cloudCanvas);
  const cloudMat = new THREE.MeshPhongMaterial({
    map: cloudTex, transparent: true, opacity: 0.7,
    depthWrite: false,
  });
  clouds = new THREE.Mesh(new THREE.SphereGeometry(R * 1.008, 64, 64), cloudMat);
  earthGroup.add(clouds);

  // Atmosphere glow
  const atmMat = new THREE.ShaderMaterial({
    uniforms: {
      c:   { value: 0.45 },
      p:   { value: 5.5 },
      glowColor: { value: new THREE.Color(0.15, 0.55, 1.0) },
      viewVector: { value: camera.position },
    },
    vertexShader: `
      uniform vec3 viewVector;
      uniform float c, p;
      varying float intensity;
      void main() {
        vec3 vNormal = normalize(normalMatrix * normal);
        vec3 vNormel = normalize(normalMatrix * viewVector);
        intensity = pow(c - dot(vNormal, vNormel), p);
        gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
      }
    `,
    fragmentShader: `
      uniform vec3 glowColor;
      varying float intensity;
      void main() {
        vec3 glow = glowColor * intensity;
        gl_FragColor = vec4(glow, intensity * 0.8);
      }
    `,
    side: THREE.FrontSide,
    blending: THREE.AdditiveBlending,
    transparent: true,
    depthWrite: false,
  });
  atmosphere = new THREE.Mesh(new THREE.SphereGeometry(R * 1.06, 64, 64), atmMat);
  earthGroup.add(atmosphere);

  // City lights (night side glow)
  const nightCanvas = document.createElement('canvas');
  nightCanvas.width = 2048; nightCanvas.height = 1024;
  const nc = nightCanvas.getContext('2d');
  drawCityLights(nc, 2048, 1024);
  const nightTex = new THREE.CanvasTexture(nightCanvas);
  const nightMat = new THREE.MeshBasicMaterial({
    map: nightTex, transparent: true, opacity: 0.6,
    blending: THREE.AdditiveBlending, depthWrite: false,
  });
  const nightMesh = new THREE.Mesh(new THREE.SphereGeometry(R * 1.001, 64, 64), nightMat);
  nightMesh.name = 'nightLights';
  earthGroup.add(nightMesh);
}

// ── PROCEDURAL TEXTURES ───────────────────────────────────────────
function drawProceduralEarth(ctx, W, H) {
  // Ocean base
  ctx.fillStyle = '#1a4a7a';
  ctx.fillRect(0, 0, W, H);

  // Deep ocean variation
  for (let i = 0; i < 800; i++) {
    const x = Math.random()*W, y = Math.random()*H;
    const r = Math.random()*40+10;
    ctx.fillStyle = `rgba(${Math.floor(Math.random()*20)},${Math.floor(60+Math.random()*40)},${Math.floor(100+Math.random()*50)},0.3)`;
    ctx.beginPath(); ctx.arc(x,y,r,0,Math.PI*2); ctx.fill();
  }

  // Continents (simplified shapes)
  ctx.fillStyle = '#3d7a3a';

  // North America
  drawBlob(ctx, 0.15*W, 0.25*H, 0.18*W, 0.2*H, '#4a8a40');
  // South America
  drawBlob(ctx, 0.22*W, 0.55*H, 0.1*W, 0.22*H, '#3d7a3a');
  // Europe
  drawBlob(ctx, 0.47*W, 0.2*H, 0.08*W, 0.12*H, '#4a8a40');
  // Africa
  drawBlob(ctx, 0.49*W, 0.42*H, 0.1*W, 0.22*H, '#5a8a30');
  // Asia
  drawBlob(ctx, 0.62*W, 0.22*H, 0.25*W, 0.3*H, '#4a8040');
  // Australia
  drawBlob(ctx, 0.76*W, 0.62*H, 0.1*W, 0.1*H, '#7a8a30');
  // Antarctica
  ctx.fillStyle = '#ddeeff';
  ctx.fillRect(0, 0.88*H, W, 0.12*H);
  ctx.fillRect(0, 0, W, 0.05*H);

  // Greenland
  drawBlob(ctx, 0.25*W, 0.08*H, 0.06*W, 0.08*H, '#ccddee');

  // Desert regions
  drawDeserts(ctx, W, H);

  // Mountains (darker areas)
  drawMountains(ctx, W, H);
}

function drawBlob(ctx, cx, cy, rx, ry, color) {
  ctx.fillStyle = color;
  ctx.beginPath();
  ctx.ellipse(cx, cy, rx, ry, 0, 0, Math.PI*2);
  ctx.fill();
  // Add rougher coastline
  for (let i = 0; i < 30; i++) {
    const angle = Math.random()*Math.PI*2;
    const xo = cx + (rx + Math.random()*rx*0.3 - rx*0.15)*Math.cos(angle);
    const yo = cy + (ry + Math.random()*ry*0.3 - ry*0.15)*Math.sin(angle);
    ctx.beginPath();
    ctx.arc(xo, yo, Math.random()*20+5, 0, Math.PI*2);
    ctx.fill();
  }
}

function drawClouds(ctx, W, H) {
  ctx.clearRect(0,0,W,H);
  for (let i = 0; i < 2000; i++) {
    const x = Math.random()*W, y = Math.random()*H;
    const r = Math.random()*60+10;
    const a = Math.random()*0.4;
    ctx.fillStyle = `rgba(255,255,255,${a})`;
    ctx.beginPath();
    ctx.ellipse(x, y, r, r*0.5, Math.random()*Math.PI, 0, Math.PI*2);
    ctx.fill();
  }
  // Storm systems
  for (let i = 0; i < 5; i++) {
    const x = Math.random()*W, y = 0.2*H + Math.random()*0.6*H;
    ctx.fillStyle = 'rgba(255,255,255,0.6)';
    ctx.beginPath();
    ctx.arc(x, y, 30+Math.random()*50, 0, Math.PI*2);
    ctx.fill();
  }
}

// ── CAMERA CONTROL ────────────────────────────────────────────────
function updateCamera(dt) {
  // Smooth lerp to targets
  const lerpSpeed = 2.5 * dt;
  camState.distance = lerp(camState.distance, camState.targetDist, lerpSpeed);
  camState.lat = lerp(camState.lat, camState.targetLat, lerpSpeed);
  camState.lon = lerpAngle(camState.lon, camState.targetLon, lerpSpeed);

  // Apply inertia to velocity
  camState.velocity.lon *= 0.92;
  camState.velocity.lat *= 0.92;
  if (!camState.isDragging) {
    camState.targetLon += camState.velocity.lon;
    camState.targetLat += camState.velocity.lat;
  }

  // Clamp latitude
  camState.targetLat = Math.max(-89, Math.min(89, camState.targetLat));

  // Convert spherical to cartesian
  const phi   = THREE.MathUtils.degToRad(90 - camState.lat);
  const theta = THREE.MathUtils.degToRad(camState.lon);
  const r     = camState.distance;

  camera.position.set(
    r * Math.sin(phi) * Math.cos(theta),
    r * Math.cos(phi),
    r * Math.sin(phi) * Math.sin(theta),
  );
  camera.lookAt(0, 0, 0);

  // Update atmosphere shader
  if (atmosphere && atmosphere.material.uniforms) {
    atmosphere.material.uniforms.viewVector.value = camera.position;
  }

  // Update HUD zoom
  const zoomLevel = (8 / camState.distance).toFixed(1);
  document.getElementById('hud-zoom-val').textContent = `ZOOM  ${zoomLevel}×`;

  // Determine mode
  let mode = 'ORBITAL';
  if (camState.distance < 0.5)      mode = 'STREET';
  else if (camState.distance < 1.5) mode = 'CITY';
  else if (camState.distance < 3)   mode = 'COUNTRY';
  else if (camState.distance < 5)   mode = 'CONTINENT';
  camState.mode = mode;
  document.getElementById('hud-mode-val').textContent = mode;

  // Lat/lon from camera direction
  const camDir = new THREE.Vector3();
  camDir.subVectors(new THREE.Vector3(0,0,0), camera.position).normalize();
  const latVal = (90 - THREE.MathUtils.radToDeg(Math.acos(camDir.y))).toFixed(2);
  const lonVal = (THREE.MathUtils.radToDeg(Math.atan2(camDir.z, camDir.x))).toFixed(2);
  document.getElementById('hud-lat').textContent = `LAT  ${latVal}°`;
  document.getElementById('hud-lon').textContent = `LON  ${lonVal}°`;
  document.getElementById('hud-alt').textContent = `ALT  ${((camState.distance-1)*6371).toFixed(0)} km`;
}

function lerp(a, b, t) { return a + (b - a) * Math.min(t, 1); }
function lerpAngle(a, b, t) {
  let diff = b - a;
  while (diff > 180) diff -= 360;
  while (diff < -180) diff += 360;
  return a + diff * Math.min(t, 1);
}

// ── MOUSE CONTROLS ────────────────────────────────────────────────
function initMouseControls() {
  const canvas = renderer.domElement;
  canvas.addEventListener('mousedown', e => {
    camState.isDragging = true;
    camState.lastMouse = { x: e.clientX, y: e.clientY };
    camState.velocity = { lon: 0, lat: 0 };
  });
  canvas.addEventListener('mousemove', e => {
    if (!camState.isDragging) return;
    const dx = e.clientX - camState.lastMouse.x;
    const dy = e.clientY - camState.lastMouse.y;
    const sensitivity = camState.distance * 0.15;
    camState.targetLon -= dx * sensitivity;
    camState.targetLat += dy * sensitivity;
    camState.velocity.lon = -dx * sensitivity * 0.4;
    camState.velocity.lat =  dy * sensitivity * 0.4;
    camState.lastMouse = { x: e.clientX, y: e.clientY };
  });
  canvas.addEventListener('mouseup', () => { camState.isDragging = false; });
  canvas.addEventListener('mouseleave', () => { camState.isDragging = false; });
  canvas.addEventListener('wheel', e => {
    const factor = e.deltaY > 0 ? 1.12 : 0.89;
    camState.targetDist = Math.max(1.08, Math.min(18, camState.targetDist * factor));
    e.preventDefault();
  }, { passive: false });

  // Click to query location
  canvas.addEventListener('click', e => {
    if (camState.isDragging) return;
    const loc = screenToGeo(e.clientX, e.clientY);
    if (loc) showLocationInfo(loc, e.clientX, e.clientY);
  });
}

function screenToGeo(sx, sy) {
  const rect = renderer.domElement.getBoundingClientRect();
  const x = ((sx - rect.left) / rect.width)  * 2 - 1;
  const y = -((sy - rect.top)  / rect.height) * 2 + 1;
  const raycaster = new THREE.Raycaster();
  raycaster.setFromCamera({ x, y }, camera);
  const hits = raycaster.intersectObject(earth);
  if (hits.length === 0) return null;
  const pt = hits[0].point.normalize();
  const lat = THREE.MathUtils.radToDeg(Math.asin(pt.y));
  const lon = THREE.MathUtils.radToDeg(Math.atan2(pt.z, pt.x));
  return { lat, lon };
}

function showLocationInfo(loc, sx, sy) {
  const popup = document.getElementById('location-popup');
  document.getElementById('pop-title').textContent =
    `${loc.lat.toFixed(4)}°, ${loc.lon.toFixed(4)}°`;

  popup.style.display = 'block';
  popup.style.left = Math.min(sx + 16, window.innerWidth - 230) + 'px';

  // Reticle
  const r = document.getElementById('reticle');
  r.style.display = 'block';
  r.style.left = sx + 'px';
  r.style.top  = sy + 'px';

  setTimeout(() => {
    popup.style.display = 'none';
    r.style.display = 'none';
  }, 3000);
}

// ── TOUCH CONTROLS ────────────────────────────────────────────────
function initTouchControls() {
  const canvas = renderer.domElement;
  let lastTouches = [];
  canvas.addEventListener('touchstart', e => {
    lastTouches = Array.from(e.touches);
    camState.velocity = { lon: 0, lat: 0 };
  });
  canvas.addEventListener('touchmove', e => {
    e.preventDefault();
    const touches = Array.from(e.touches);
    if (touches.length === 1) {
      const dx = touches[0].clientX - lastTouches[0].clientX;
      const dy = touches[0].clientY - lastTouches[0].clientY;
      const s  = camState.distance * 0.12;
      camState.targetLon -= dx * s;
      camState.targetLat += dy * s;
    } else if (touches.length === 2 && lastTouches.length === 2) {
      const d0 = dist2D(lastTouches[0], lastTouches[1]);
      const d1 = dist2D(touches[0], touches[1]);
      if (d0 > 0) {
        const factor = d0 / d1;
        camState.targetDist = Math.max(1.08, Math.min(18, camState.targetDist * factor));
      }
    }
    lastTouches = touches;
  }, { passive: false });
}

function dist2D(a, b) {
  return Math.hypot(a.clientX - b.clientX, a.clientY - b.clientY);
}

// ── HAND GESTURE CONTROL ─────────────────────────────────────────
async function initHandGestures() {
  try {
    const stream = await navigator.mediaDevices.getUserMedia({ video: true });
    const video  = document.getElementById('cam-video');
    video.srcObject = stream;
    document.getElementById('cam-preview').style.display = 'block';

    // Load MediaPipe via CDN
    const script1 = document.createElement('script');
    script1.src = 'https://cdn.jsdelivr.net/npm/@mediapipe/hands/hands.js';
    document.head.appendChild(script1);
    script1.onload = () => {
      const script2 = document.createElement('script');
      script2.src = 'https://cdn.jsdelivr.net/npm/@mediapipe/camera_utils/camera_utils.js';
      document.head.appendChild(script2);
      script2.onload = () => startHandTracking(video);
    };
  } catch(e) {
    document.getElementById('gesture-status').textContent = 'HAND TRACKING: UNAVAILABLE';
    document.getElementById('hud-gesture').textContent = '● GESTURE  MOUSE FALLBACK';
  }
}

function startHandTracking(video) {
  if (!window.Hands) return;
  const hands = new window.Hands({
    locateFile: f => `https://cdn.jsdelivr.net/npm/@mediapipe/hands/${f}`
  });
  hands.setOptions({
    maxNumHands: 2,
    modelComplexity: 1,
    minDetectionConfidence: 0.7,
    minTrackingConfidence: 0.6,
  });
  hands.onResults(onHandResults);

  const camera = new window.Camera(video, {
    onFrame: async () => { await hands.send({ image: video }); },
    width: 320, height: 240,
  });
  camera.start();
  document.getElementById('hud-gesture').textContent = '● GESTURE  ACTIVE';
}

let prevHandData = null;

function onHandResults(results) {
  const hands = results.multiHandLandmarks || [];
  if (hands.length === 0) {
    gestureState.prevDist = null;
    prevHandData = null;
    return;
  }

  const W = window.innerWidth, H = window.innerHeight;

  if (hands.length === 2) {
    // Two hand: pinch-zoom + orbit
    const h0 = hands[0], h1 = hands[1];
    const midX = (h0[9].x + h1[9].x) * 0.5;
    const midY = (h0[9].y + h1[9].y) * 0.5;
    const d    = Math.hypot(h0[9].x - h1[9].x, h0[9].y - h1[9].y);

    if (gestureState.prevDist !== null) {
      // Zoom: hands spreading = zoom in (lower distance from Earth)
      const dd = d - gestureState.prevDist;
      camState.targetDist = Math.max(1.08, Math.min(18,
        camState.targetDist - dd * 6));
      // Pan from mid-point
      if (gestureState.prevMidX !== null) {
        const dxn = midX - gestureState.prevMidX;
        const dyn = midY - gestureState.prevMidY;
        camState.targetLon -= dxn * 60;
        camState.targetLat += dyn * 40;
      }
    }
    gestureState.prevDist = d;
    gestureState.prevMidX = midX;
    gestureState.prevMidY = midY;

    // Check pinch for selection
    const pinch0 = Math.hypot(h0[4].x - h0[8].x, h0[4].y - h0[8].y) < 0.04;
    const pinch1 = Math.hypot(h1[4].x - h1[8].x, h1[4].y - h1[8].y) < 0.04;
    if ((pinch0 || pinch1) && !gestureState.pinching) {
      gestureState.pinching = true;
      const h = pinch0 ? h0 : h1;
      const sx = (1 - h[8].x) * W;
      const sy = h[8].y * H;
      const loc = screenToGeo(sx, sy);
      if (loc) showLocationInfo(loc, sx, sy);
      updateHUDGesture('PINCH SELECT');
    } else if (!pinch0 && !pinch1) {
      gestureState.pinching = false;
    }
  } else {
    // One hand: orbit
    gestureState.prevDist = null;
    gestureState.prevMidX = null;
    const h = hands[0];
    if (prevHandData) {
      const dx = h[9].x - prevHandData[9].x;
      const dy = h[9].y - prevHandData[9].y;
      camState.targetLon -= dx * 80;
      camState.targetLat += dy * 50;
      camState.velocity.lon = -dx * 20;
      camState.velocity.lat =  dy * 15;
    }
    prevHandData = h;

    const isPinching = Math.hypot(h[4].x - h[8].x, h[4].y - h[8].y) < 0.04;
    if (isPinching) updateHUDGesture('PINCH GRAB');
    else            updateHUDGesture('OPEN HAND');
  }
}

function updateHUDGesture(text) {
  document.getElementById('gesture-status').textContent = text;
}

// ── VOICE COMMANDS ────────────────────────────────────────────────
function flyTo(key) {
  const loc = LOCATIONS[key.toLowerCase()];
  if (!loc) { showVoice(`LOCATION NOT FOUND: ${key.toUpperCase()}`); return; }
  showVoice(`NAVIGATING TO ${loc.name.toUpperCase()}`);
  // First zoom out slightly, then zoom in to location
  const prevDist = camState.targetDist;
  camState.targetDist = Math.max(camState.targetDist, 5);
  setTimeout(() => {
    camState.targetLat  = loc.lat;
    camState.targetLon  = loc.lon;
    camState.targetDist = loc.zoom * 2;
    document.getElementById('hud-loc-name').textContent = loc.name.toUpperCase();
    document.getElementById('hud-loc-sub').textContent  = loc.sub;
  }, 800);
}

function setMode(mode) {
  switch(mode) {
    case 'satellite':
      showVoice('SATELLITE IMAGERY MODE');
      document.getElementById('hud-data-src').textContent = '● SOURCE   SATELLITE MOSAIC';
      break;
    case 'night':
      showVoice('EARTH AT NIGHT — CITY LIGHTS');
      const nightMesh = earthGroup.getObjectByName('nightLights');
      if (nightMesh) nightMesh.material.opacity = 1.0;
      sunLight.intensity = 0.2;
      document.getElementById('hud-data-src').textContent = '● SOURCE   NASA NIGHT LIGHTS';
      break;
    case 'day':
      showVoice('DAYTIME MODE');
      const nightM = earthGroup.getObjectByName('nightLights');
      if (nightM) nightM.material.opacity = 0.5;
      sunLight.intensity = 3.5;
      break;
    case 'reset':
      camState.targetDist = 6;
      camState.targetLat  = 20;
      camState.targetLon  = 0;
      showVoice('RESETTING VIEW TO ORBITAL');
      document.getElementById('hud-loc-name').textContent = 'ORBITAL VIEW';
      document.getElementById('hud-loc-sub').textContent  = 'Earth · Altitude 31,850 km';
      break;
  }
}

function showVoice(text) {

  el.classList.add('show');
  clearTimeout(showVoice._t);
  showVoice._t = setTimeout(() => el.classList.remove('show'), 3000);
}

// Expose for Python bridge
window.earthCommand = function(cmd) {
  const c = cmd.toLowerCase().trim();
  if (c.includes('show me ') || c.includes('take me to ') ||
      c.includes('go to ') || c.includes('find ') || c.includes('navigate to ')) {
    const target = c.replace(/show me|take me to|go to|find|navigate to/g,'').trim();
    flyTo(target);
  } else if (c.includes('satellite mode') || c.includes('satellite')) {
    setMode('satellite');
  } else if (c.includes('night') || c.includes('city lights')) {
    setMode('night');
  } else if (c.includes('day mode') || c.includes('daytime')) {
    setMode('day');
  } else if (c.includes('reset') || c.includes('orbital')) {
    setMode('reset');
  } else if (c.includes('my location')) {
    navigator.geolocation.getCurrentPosition(pos => {
      camState.targetLat  = pos.coords.latitude;
      camState.targetLon  = pos.coords.longitude;
      camState.targetDist = 1.5;
      showVoice(`MY LOCATION: ${pos.coords.latitude.toFixed(2)}°, ${pos.coords.longitude.toFixed(2)}°`);
    });
  } else if (c.includes('exit') || c.includes('close')) {
    exitEarthMode();
  }
};

// ── ANIMATION LOOP ────────────────────────────────────────────────
let frameCount = 0, fpsTime = performance.now(), fps = 60;

function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.1);
  frameCount++;
  const now = performance.now();
  if (now - fpsTime >= 1000) {
    fps = frameCount;
    frameCount = 0;
    fpsTime = now;
    document.getElementById('hud-fps').textContent = `${fps} FPS`;
  }

  if (earth)  earth.rotation.y  += 0.0005;
  if (clouds) clouds.rotation.y += 0.0006;
  if (stars)  stars.rotation.y  += 0.00002;

  updateCamera(dt);
  renderer.render(scene, camera);
}
